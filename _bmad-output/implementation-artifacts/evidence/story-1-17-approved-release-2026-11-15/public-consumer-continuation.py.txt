source='f792bba6c0564d3c36dda6ad1c2fc55b0b0a6331'

projects=work/'references/Hexalith.Projects'

old_env={s.split('=',1)[0]:s.split('=',1)[1] for s in Path('/proc/3021582/environ').read_bytes().decode().split('\0') if '=' in s}

    observed=subprocess.check_output(['git','diff','HEAD','--binary'],cwd=owner)

    args=['pwsh','-NoLogo','-NoProfile','-File','Tools/'+script]
    log=evidence['Builds']/(script+'.telemetry-disabled.log')
    assert not log.exists()
    print('START',script,flush=True);started=time.monotonic()
    with log.open('w') as f:
        f.write(json.dumps(args)+'\nDOTNET_CLI_TELEMETRY_OPTOUT=1\n');f.flush()

bindings=[]
for owner,cwd,project in [("folders",work,"Hexalith.Folders.Server.Tests"),("folders",work,"Hexalith.Folders.Client.Tests"),("folders",work,"Hexalith.Folders.IntegrationTests"),("folders",work,"Hexalith.Folders.Cli.Tests"),("folders",work,"Hexalith.Folders.Mcp.Tests"),("Projects",projects,"Hexalith.Projects.Server.Tests")]:
    directory=cwd/"tests"/project/"bin/Release/net11.0";deps=json.loads((directory/(project+".deps.json")).read_text())
    target=next(v for k,v in deps["targets"].items() if k.startswith(".NETCoreApp,Version=v11.0"))
    for identity,item in target.items():
        if identity not in archives:continue
        archive=archives[identity]
        for member in item.get("runtime",{}):
            filename=Path(member).name
            if not filename.endswith(".dll"):continue
            runtime=directory/filename;assert runtime.is_file()
            with zipfile.ZipFile(archive) as z:expected=hashlib.sha256(z.read(member)).hexdigest()
            assert digest(runtime)==expected,(identity,runtime)
            bindings.append({"project":project,"package":identity,"member":member,"archive_sha256":digest(archive),"runtime_sha256":expected})
    if project.endswith("Server.Tests"):
        selected={r["package"] for r in bindings if r["project"]==project}
        required={f"Hexalith.EventStore.{s}/5.1.1" for s in ["Client","Contracts","Server"]}

        assert required<=selected,(required-selected)
    with tarfile.open(evidence[owner]/(project+"-runtime.tar.gz"),"w:gz") as t:t.add(directory,arcname=project)
